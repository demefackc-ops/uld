    <div class="max-w-[1600px] mx-auto">
        <header class="mb-8 flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
            <div>
                <h1 class="text-2xl font-extrabold text-slate-900">Course Development Pulse</h1>
                <p class="text-slate-500 text-sm">Live Sync: Google Sheets</p>
            </div>
            <div class="flex items-center gap-4">
                <input type="text" id="search" placeholder="Search courses or IDs..." class="border border-slate-200 rounded-lg px-4 py-2 text-sm focus:ring-2 focus:ring-blue-500 outline-none w-64">
                <button onclick="fetchData()" class="bg-slate-900 text-white px-4 py-2 rounded-lg text-sm font-bold hover:bg-slate-800 transition">Refresh</button>
            </div>
        </header>

        <!-- Stats Grid -->
        <div class="grid grid-cols-1 md:grid-cols-4 gap-4 mb-8">
            <div class="card p-4 border-l-4 border-blue-500">
                <div class="text-xs font-bold text-slate-400 uppercase tracking-wider">Total Projects</div>
                <div class="text-2xl font-bold" id="stat-total">0</div>
            </div>
            <div class="card p-4 border-l-4 border-emerald-500">
                <div class="text-xs font-bold text-slate-400 uppercase tracking-wider">Avg. Progress</div>
                <div class="text-2xl font-bold" id="stat-avg">0%</div>
            </div>
            <div class="card p-4 border-l-4 border-amber-500">
                <div class="text-xs font-bold text-slate-400 uppercase tracking-wider">New Courses</div>
                <div class="text-2xl font-bold" id="stat-new">0</div>
            </div>
            <div class="card p-4 border-l-4 border-purple-500">
                <div class="text-xs font-bold text-slate-400 uppercase tracking-wider">Redesigns</div>
                <div class="text-2xl font-bold" id="stat-redesign">0</div>
            </div>
        </div>

        <!-- Charts -->
        <div class="grid grid-cols-1 lg:grid-cols-3 gap-6 mb-8">
            <div class="card p-6 lg:col-span-2 h-80">

                <canvas id="typeChart"></canvas>
            </div>
        </div>

        <!-- Table -->

            <div class="overflow-x-auto">
                <table class="w-full text-left border-collapse">
                    <thead class="bg-slate-50 border-b border-slate-200">
                        <tr class="text-[10px] uppercase tracking-widest text-slate-400 font-bold">
                            <th class="p-4">Designer</th>

    <script>
        const SHEET_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vS7ufpgzKAtcaZPRYLWQ-5Z8uVajX35irELgE37yrxkVY9hnIRYAqxakh7qslwVONWrsnrdJ_L8nZ2l/pubhtml';
        let allData = [];
        let workloadChart, typeChart;

        async function fetchData() {
            document.getElementById('loading-overlay').style.display = 'flex';
            Papa.parse(SHEET_URL, {
                download: true,
                header: true,
                complete: function(results) {
                    allData = results.data.filter(row => row.Designer);
                    updateUI(allData);
                    document.getElementById('loading-overlay').style.display = 'none';
                }
            });
        }

        function updateUI(data) {
            renderStats(data);
            renderTable(data);
            renderCharts(data);
        }

        function renderStats(data) {
            const total = data.length;
            const avg = Math.round(data.reduce((acc, curr) => acc + parseInt(curr['Percent Complete'] || 0), 0) / total);
            const news = data.filter(d => d.Type === 'New Course').length;
            const redesigns = data.filter(d => d.Type === 'Redesign').length;

            document.getElementById('stat-total').innerText = total;
            document.getElementById('stat-avg').innerText = avg + '%';
            document.getElementById('stat-new').innerText = news;
            document.getElementById('stat-redesign').innerText = redesigns;
        }

        function renderTable(data) {
            const tbody = document.getElementById('table-body');
            tbody.innerHTML = data.map(row => `
                <tr class="hover:bg-slate-50 transition-colors">
                    <td class="p-4 font-bold text-blue-600">${row.Designer}</td>
                    <td class="p-4 text-xs text-slate-500">${row['Development Term']}</td>
                    <td class="p-4 text-xs font-medium">${row.Project}</td>
                    <td class="p-4">
                        <div class="font-bold text-slate-800">${row['Course ID']}</div>
                        <div class="text-xs text-slate-500">${row['Course Title']}</div>
                    </td>
                    <td class="p-4">
                        <span class="px-2 py-1 rounded text-[10px] font-bold uppercase ${row.Type === 'New Course' ? 'bg-blue-100 text-blue-700' : 'bg-amber-100 text-amber-700'}">
                            ${row.Type}
                        </span>
                    </td>
                    <td class="p-4 w-40">
                        <div class="flex items-center gap-2">
                            <div class="flex-1 h-1.5 bg-slate-100 rounded-full overflow-hidden">
                                <div class="h-full bg-blue-500" style="width: ${row['Percent Complete']}"></div>
                            </div>
                            <span class="text-[10px] font-bold">${row['Percent Complete']}</span>
                        </div>
                    </td>
                </tr>
            `).join('');
        }

        function renderCharts(data) {
            const designers = {};
            const types = { 'New Course': 0, 'Redesign': 0 };
            data.forEach(d => {
                designers[d.Designer] = (designers[d.Designer] || 0) + 1;
                if(types[d.Type] !== undefined) types[d.Type]++;
            });

            if(workloadChart) workloadChart.destroy();
            if(typeChart) typeChart.destroy();

            workloadChart = new Chart(document.getElementById('workloadChart'), {
                type: 'bar',
                data: {
                    labels: Object.keys(designers),
                    datasets: [{ label: 'Courses', data: Object.values(designers), backgroundColor: '#3b82f6', borderRadius: 4 }]
                },
                options: { maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true, grid: { display: false } } } }
            });

            typeChart = new Chart(document.getElementById('typeChart'), {
                type: 'doughnut',
                data: {
                    labels: Object.keys(types),
                    datasets: [{ data: Object.values(types), backgroundColor: ['#3b82f6', '#f59e0b'], borderWidth: 0 }]
                },
                options: { maintainAspectRatio: false, cutout: '75%', plugins: { legend: { position: 'bottom' } } }
            });
        }

        document.getElementById('search').addEventListener('input', (e) => {
            const term = e.target.value.toLowerCase();
            const filtered = allData.filter(d => 
                d.Designer.toLowerCase().includes(term) || 
                d['Course Title'].toLowerCase().includes(term) ||
                d['Course ID'].toLowerCase().includes(term)
            );
            renderTable(filtered);
        });

        fetchData();
    </script>
